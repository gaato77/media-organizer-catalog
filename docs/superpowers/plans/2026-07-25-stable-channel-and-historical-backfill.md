# Stable Catalog Channel and Historical Backfill Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Publish durable base, supplement, and current-year catalog components plus a strict `catalog/channel/stable.json` contract that Media Organizer can install and update automatically.

**Architecture:** Keep each catalog component as an independently versioned GitHub Release containing the existing validated `catalog-full.zip` and `manifest.json`. Add committed component pointers with package and installed-database hashes, then assemble those pointers into one atomic stable channel document. Reuse the existing year probe, consolidation, package validation, and current-year workflows instead of rebuilding those systems.

**Tech Stack:** Python 3.12, dataclasses, JSON, SQLite 3, pytest, Ruff, mypy, GitHub Actions, GitHub CLI, existing `media_catalog_builder` package.

## Global Constraints

- The public repository is the unauthenticated distribution origin.
- Releases, not Actions artifacts, are the durable package source.
- Existing catalog schema remains version 1.
- Existing release manifest schema remains version 1.
- Every component pointer contains package size/SHA-256 and installed SQLite size/SHA-256.
- `stable.json` must never reference an unpublished or unvalidated release.
- Component year ranges may overlap only when priorities differ; higher priority wins.
- Component types are exactly `base`, `supplement`, `previous-year`, and `current-year`.
- The stable channel must not hardcode a permanent current year.
- Historical base and supplement are immutable once published under a versioned release tag.
- Scheduled current-year publication updates the current pointer and stable channel only after release upload succeeds.
- The initial application path uses full packages; delta generation is outside this plan.
- CI validates JSON contracts, release assets, SQLite integrity, representative lookups, workflow structure, and year coverage.

```python
from __future__ import annotations

import json
import os
import re
from dataclasses import dataclass
from datetime import UTC, datetime
from enum import StrEnum
from pathlib import Path
from typing import Any, cast

_SHA256 = re.compile(r"^[0-9a-f]{64}$")
_VERSION = re.compile(r"^[0-9]{4}\.[0-9]{2}\.[0-9]{2}$")
_SAFE_NAME = re.compile(r"^[^/\\\x00]+$")
type JsonObject = dict[str, Any]


class ComponentType(StrEnum):
    BASE = "base"
    SUPPLEMENT = "supplement"
    PREVIOUS_YEAR = "previous-year"
    CURRENT_YEAR = "current-year"


@dataclass(frozen=True, slots=True)
class CatalogComponent:
    id: str
    type: ComponentType
    from_year: int
    to_year: int
    version: str
    release_tag: str
    manifest_asset: str
    package_name: str
    package_bytes: int
    package_sha256: str
    installed_name: str
    installed_bytes: int
    installed_sha256: str
    catalog_schema: int
    minimum_app_version: str
    priority: int

    def __post_init__(self) -> None:
        if not self.id or self.id != self.id.strip() or "/" in self.id or "\\" in self.id:
            raise ValueError("invalid component id")
        if not 1800 <= self.from_year <= self.to_year <= 2200:
            raise ValueError("invalid component year range")
        if _VERSION.fullmatch(self.version) is None:
            raise ValueError("invalid component version")
        for label, value in (("manifest", self.manifest_asset), ("package", self.package_name), ("installed", self.installed_name)):
            if _SAFE_NAME.fullmatch(value) is None or Path(value).name != value:
                raise ValueError(f"invalid {label} name")
        if self.package_bytes <= 0 or self.installed_bytes <= 0:
            raise ValueError("component sizes must be positive")
        if _SHA256.fullmatch(self.package_sha256) is None or _SHA256.fullmatch(self.installed_sha256) is None:
            raise ValueError("component SHA-256 must contain 64 lowercase hexadecimal characters")
        if self.catalog_schema != 1:
            raise ValueError("catalog schema must be 1")
        if not self.minimum_app_version.strip():
            raise ValueError("minimum app version is required")

    def to_dict(self) -> JsonObject:
        return {
            "id": self.id,
            "type": self.type.value,
            "from_year": self.from_year,
            "to_year": self.to_year,
            "version": self.version,
            "release_tag": self.release_tag,
            "manifest_asset": self.manifest_asset,
            "package_name": self.package_name,
            "package_bytes": self.package_bytes,
            "package_sha256": self.package_sha256,
            "installed_name": self.installed_name,
            "installed_bytes": self.installed_bytes,
            "installed_sha256": self.installed_sha256,
            "catalog_schema": self.catalog_schema,
            "minimum_app_version": self.minimum_app_version,
            "priority": self.priority,
        }
```

Implement `from_dict` with an exact field set. `StableChannel.__post_init__` must enforce schema 1, channel `stable`, UTC `published_at`, unique IDs, and reject overlapping ranges at the same priority. Store components sorted by `(-priority, id)` using `object.__setattr__`.

```python
from __future__ import annotations

from pathlib import Path

from media_catalog_builder.channel import CatalogComponent, ComponentType
from media_catalog_builder.manifest import load_manifest
from media_catalog_builder.package import sha256_file


def build_component_pointer(
    release_dir: Path,
    catalog_path: Path,
    *,
    component_id: str,
    component_type: str,
    from_year: int,
    to_year: int,
    release_tag: str,
    priority: int,
) -> CatalogComponent:
    manifest = load_manifest(release_dir / "manifest.json")
    if not catalog_path.is_file():
        raise ValueError("installed catalog is missing")
    return CatalogComponent(
        id=component_id,
        type=ComponentType(component_type),
        from_year=from_year,
        to_year=to_year,
        version=manifest.catalog_version,
        release_tag=release_tag,
        manifest_asset="manifest.json",
        package_name=manifest.full.name,
        package_bytes=manifest.full.download_bytes,
        package_sha256=manifest.full.sha256,
        installed_name="catalog.sqlite",
        installed_bytes=catalog_path.stat().st_size,
        installed_sha256=sha256_file(catalog_path),
        catalog_schema=manifest.catalog_schema,
        minimum_app_version=manifest.minimum_app_version,
        priority=priority,
    )
```

The CLI loads arguments, calls this function, and uses `write_component_atomic`.

from __future__ import annotations

import argparse
from datetime import datetime
from pathlib import Path

from media_catalog_builder.channel import StableChannel, load_component, write_stable_channel_atomic


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--components-dir", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--published-at", required=True)
    args = parser.parse_args()
    published_at = datetime.fromisoformat(args.published_at.replace("Z", "+00:00"))
    components = tuple(load_component(path) for path in sorted(args.components_dir.glob("*.json")))
    if not components:
        raise ValueError("stable channel requires at least one component")
    channel = StableChannel(1, "stable", published_at, components)
    write_stable_channel_atomic(args.output, channel)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
